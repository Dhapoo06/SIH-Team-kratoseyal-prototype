from flask import Flask, render_template, request, jsonify
import sqlite3, hashlib

app = Flask(__name__)

# Team Kratoseyal.. — SIH26261 AICTE Verification Engine
APH_NORMS = {
    "min_pcs": 30,
    "max_student_faculty_ratio": 20,
    "min_land_acres": 2.5,
    "min_builtup_sqm": 4200,
    "min_fdr_lakhs": 15.0
}

@app.route('/')
def index():
    return render_template('index.html')

@app.route('/api/verify', methods=['POST'])
def verify_dossier():
    payload = request.get_json(silent=True) or {}
    pcs = int(payload.get("pcs", 36))
    ratio = int(payload.get("ratio", 18))
    land = float(payload.get("land", 3.2))
    fdr = float(payload.get("fdr", 15.0))

    violations = []
    if pcs < APH_NORMS["min_pcs"]:
        violations.append("CSE Lab PC shortfall")
    if ratio > APH_NORMS["max_student_faculty_ratio"]:
        violations.append("Faculty-Student ratio non-compliant")
    if land < APH_NORMS["min_land_acres"]:
        violations.append("Contiguous land below 2.5 acres")
    if fdr < APH_NORMS["min_fdr_lakhs"]:
        violations.append("FDR security deposit shortfall")

    return jsonify({
        "team": "Team Kratoseyal..",
        "problem_code": "SIH26261",
        "compliant": len(violations) == 0,
        "violations": violations
    })

if __name__ == '__main__':
    app.run(host='127.0.0.1', port=5000, debug=True)
